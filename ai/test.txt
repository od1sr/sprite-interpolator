$body = @{
    input_uri  = "D:/sprite-interpolator/ai/test2.mp4"
    output_uri = "D:/sprite-interpolator/ai/test_upscaled.mp4"
    params     = @{ scale = 2; tile_size = 512 }
} | ConvertTo-Json -Compress

$sw = [System.Diagnostics.Stopwatch]::StartNew()
$result = Invoke-RestMethod -Uri "http://localhost:8001/internal/v1/upscale" `
    -Method POST -ContentType "application/json" -Body $body -TimeoutSec 9000
$sw.Stop()

Write-Host "Время: $($sw.Elapsed.TotalSeconds) сек"
$result | ConvertTo-Json -Depth 5
